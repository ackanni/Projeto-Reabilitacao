void setup() {
  Serial.begin(9600); // Mesma velocidade da sua ArduinoReaderThread
  
  pinMode(4, INPUT_PULLUP); // Botão Acelerador ligado no GPIO 4 e GND
  pinMode(5, INPUT_PULLUP); // Botão Freio ligado no GPIO 5 e GND
}

void loop() {
  // LOW significa que o botão foi pressionado (fechou o circuito com o GND)
  bool aceleradorPressionado = (digitalRead(4) == LOW);
  bool freioPressionado = (digitalRead(5) == LOW);

  if (aceleradorPressionado) {
    Serial.println("1"); // Acelerando
  } else if (freioPressionado) {
    Serial.println("2"); // Freando
  } else {
    Serial.println("0"); // Solto (Dorsiflexão)
  }

  delay(50); // Evita flood na porta serial
}
